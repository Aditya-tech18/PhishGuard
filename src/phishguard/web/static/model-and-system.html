def extract_url_vector(raw_url: str) -> np.ndarray:

                parsed = urlparse(raw_url)

                entropy = shannon_entropy(parsed.path)

                depth = parsed.netloc.count('.')

                brand_dist = min_levenshtein(parsed.netloc)

                vowel_ratio = calc_vowel_ratio(parsed.netloc)

                # 55-element fixed float32 vector

                features = np.zeros(55, dtype=np.float32)

                features[0] = entropy

                features[1] = depth

                features[2] = brand_dist

                return onnx_session.run(features)
